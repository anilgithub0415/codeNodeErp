import { Request, Response, NextFunction } from 'express';
// 💡 THIS IMPORT REFERS TO THE EXACT SAME CONNECTION POOL WE INITIALIZED
import { redisClient } from './redis-client'; 

export const cacheMiddleware = (ttl: number) => {
  return async (req: Request, res: Response, next: NextFunction): Promise<void> => {
    if (req.method !== 'GET') return next();

    const cacheKey = `express:cache:${req.originalUrl}`;

    try {
      // Accesses the open connection pool instantly without creating a new connection
      const cachedResponse = await redisClient.get(cacheKey);
      
      if (cachedResponse) {
        res.setHeader('X-Cache-Status', 'HIT');
        res.setHeader('Content-Type', 'application/json');
        res.send(cachedResponse); 
        return;
      }

      const originalSend = res.send;

      res.send = function (body: any): Response {
        res.send = originalSend;

        if (res.statusCode === 200 && body) {
          // Reuses the pool to write data asynchronously
          redisClient.setEx(cacheKey, ttl, typeof body === 'string' ? body : JSON.stringify(body))
            .catch(err => console.error('Redis write error:', err));
        }

        res.setHeader('X-Cache-Status', 'MISS');
        return originalSend.call(this, body);
      };

      next();
    } catch (error) {
      next(); // Fail safely if Redis is down
    }
  };
};

// 💡 IMPORTS THE EXACT SAME SINGLETON CONNECTION POOL
import { redisClient } from './redis-client';

/**
 * Universal Cache Invalidation engine for clearing stale server mutations.
 */
export const cacheInvalidator = {
  /**
   * Drops a singular, deterministic key from the memory grid instantly.
   * Use this for explicit resource records like '/api/customers/42'
   */
  async invalidateKey(key: string): Promise<void> {
    try {
      const cacheKey = `express:cache:${key}`;
      await redisClient.del(cacheKey);
    } catch (error) {
      console.error(`🔴 Redis Engine: Failed to evict explicit key [${key}]:`, error);
    }
  },

  /**
   * Scans and deletes all collection variations matching a base directory path.
   * This clears variants like '/api/customers', '/api/customers?status=ACTIVE', etc.
   */
  async invalidatePattern(routePattern: string): Promise<void> {
    const searchPattern = `express:cache:${routePattern}*`;
    try {
      let cursor = 0;
      
      do {
        // Use SCAN instead of KEYS to avoid blocking the single-threaded Redis thread pool
        const reply = await redisClient.scan(cursor, { 
          MATCH: searchPattern, 
          COUNT: 100 
        });
        
        cursor = reply.cursor;
        const matchingKeys = reply.keys;

        // If stale variations are detected, perform an atomic cluster deletion block
        if (matchingKeys.length > 0) {
          await redisClient.del(matchingKeys);
        }
      } while (cursor !== 0); // Terminate execution loop once memory registry pointer resets to zero
      
    } catch (error) {
      console.error(`🔴 Redis Engine: Pattern scanning execution failure [${searchPattern}]:`, error);
    }
  }
};

how to invalidate cache

/**
 * Safely evicts targeted keys or patterns from the Redis cache instance.
 */
export const cacheInvalidator = {
  /**
   * Delete an exact single key
   */
  async invalidateKey(key: string): Promise<void> {
    try {
      await redisClient.del(key);
    } catch (error) {
      console.error(`Failed to invalidate exact key: ${key}`, error);
    }
  },

  /**
   * Scan and delete all keys matching a specific resource route pattern.
   * This clears variants like /api/customers, /api/customers?page=1, etc.
   */
  async invalidatePattern(routePattern: string): Promise<void> {
    const pattern = `express:cache:${routePattern}*`;
    try {
      let cursor = 0;
      do {
        // Scan for keys matching the pattern to avoid blocking the single-threaded Redis process
        const reply = await redisClient.scan(cursor, { MATCH: pattern, COUNT: 100 });
        cursor = reply.cursor;
        const keys = reply.keys;

        if (keys.length > 0) {
          await redisClient.del(keys);
        }
      } while (cursor !== 0);
    } catch (error) {
      console.error(`Failed to invalidate pattern: ${pattern}`, error);
    }
  }
};
